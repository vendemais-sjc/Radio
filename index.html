=================================================================== # 📊 SISTEMA DE LOGS E MONITORAMENTO # =================================================================== class Logger: """Sistema profissional de logging""" def __init__(self): self.logs = [] self.start_time = time.time() def log(self, message, level="INFO"): """Registra um evento no log""" timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S") log_entry = { "timestamp": timestamp, "level": level, "message": message, "elapsed": f"{time.time() - self.start_time:.2f}s" } self.logs.append(log_entry) # Mostrar no console com cores colors = { "INFO": "\033[92m", # Verde "WARNING": "\033[93m", # Amarelo "ERROR": "\033[91m", # Vermelho "SUCCESS": "\033[95m" # Magenta } color = colors.get(level, "\033[0m") print(f"{color}[{level}] {message}\033[0m") def save_to_file(self): """Salva logs em arquivo JSON""" with open(Config.LOG_FILE, 'w') as f: json.dump(self.logs, f, indent=2) return Config.LOG_FILE def get_summary(self): """Retorna resumo dos logs""" total = len(self.logs) errors = len([l for l in self.logs if l['level'] == 'ERROR']) warnings = len([l for l in self.logs if l['level'] == 'WARNING']) return { "total_events": total, "errors": errors, "warnings": warnings, "success_rate": f"{(total - errors)/total*100:.1f}%" if total > 0 else "0%", "total_time": f"{time.time() - self.start_time:.2f}s" } # =================================================================== # 🎥 PROCESSADOR PRINCIPAL DE VÍDEO # =================================================================== class VideoProcessor: """Classe principal para processamento de vídeo""" def __init__(self, video_path, logger): self.video_path = video_path self.logger = logger self.cap = None self.info = {} self.frames_cache = {} def load_video(self): """Carrega e analisa informações do vídeo""" try: self.cap = cv2.VideoCapture(self.video_path) if not self.cap.isOpened(): raise Exception("Não foi possível abrir o vídeo") # Extrair informações detalhadas self.info = { 'width': int(self.cap.get(cv2.CAP_PROP_FRAME_WIDTH)), 'height': int(self.cap.get(cv2.CAP_PROP_FRAME_HEIGHT)), 'fps': self.cap.get(cv2.CAP_PROP_FPS), 'total_frames': int(self.cap.get(cv2.CAP_PROP_FRAME_COUNT)), 'codec': int(self.cap.get(cv2.CAP_PROP_FOURCC)), 'duration': self.cap.get(cv2.CAP_PROP_FRAME_COUNT) / self.cap.get(cv2.CAP_PROP_FPS) } # Calcular duração formatada duration_seconds = self.info['duration'] hours = int(duration_seconds // 3600) minutes = int((duration_seconds % 3600) // 60) seconds = int(duration_seconds % 60) self.info['duration_formatted'] = f"{hours:02d}:{minutes:02d}:{seconds:02d}" self.logger.log(f"Vídeo carregado: {os.path.basename(self.video_path)}", "SUCCESS") self.logger.log(f"Resolução: {self.info['width']}x{self.info['height']}", "INFO") self.logger.log(f"Duração: {self.info['duration_formatted']}", "INFO") self.logger.log(f"FPS: {self.info['fps']:.2f}", "INFO") return True except Exception as e: self.logger.log(f"Erro ao carregar vídeo: {str(e)}", "ERROR") return False def get_video_info(self): """Retorna informações detalhadas em formato amigável""" return self.info def extract_audio(self, format='mp3'): """Extrai áudio do vídeo em diferentes formatos""" self.logger.log("Iniciando extração de áudio...", "INFO") try: video_clip = VideoFileClip(self.video_path) output_file = f"{Config.OUTPUT_DIR}/audio_extracted.{format}" # Converter para formato desejado if format == 'mp3': video_clip.audio.write_audiofile(output_file, codec='mp3', bitrate='192k', verbose=False, logger=None) elif format == 'wav': video_clip.audio.write_audiofile(output_file, codec='pcm_s16le', verbose=False, logger=None) elif format == 'aac': video_clip.audio.write_audiofile(output_file, codec='aac', verbose=False, logger=None) video_clip.close() file_size = os.path.getsize(output_file) / (1024*1024) self.logger.log(f"Áudio extraído com sucesso! Tamanho: {file_size:.2f}MB", "SUCCESS") # Baixar automaticamente files.download(output_file) return output_file except Exception as e: self.logger.log(f"Erro na extração de áudio: {str(e)}", "ERROR") return None def extract_frames_advanced(self, mode='time', value=1, quality=95): """Extrai frames com múltiplos modos""" self.logger.log(f"Iniciando extração de frames (modo: {mode})...", "INFO") try: self.cap.set(cv2.CAP_PROP_POS_FRAMES, 0) frame_count = 0 saved_count = 0 metadata = [] # Determinar intervalo if mode == 'time': interval_frames = int(self.info['fps'] * value) elif mode == 'frames': interval_frames = value else: interval_frames = int(self.info['fps']) # Criar pasta para este lote timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S") frame_folder = f"{Config.FRAMES_DIR}/frames_{timestamp}" Path(frame_folder).mkdir(parents=True, exist_ok=True) # Barra de progresso total_to_extract = self.info['total_frames'] // interval_frames pbar = tqdm(total=total_to_extract, desc="Extraindo frames") while True: ret, frame = self.cap.read() if not ret: break if frame_count % interval_frames == 0: # Processar frame (melhorar qualidade) if quality < 100: encode_param = [int(cv2.IMWRITE_JPEG_QUALITY), quality] else: encode_param = [] # Salvar frame frame_filename = f"{frame_folder}/frame_{saved_count:06d}.jpg" cv2.imwrite(frame_filename, frame, encode_param) # Salvar metadados metadata.append({ 'frame_number': frame_count, 'time_seconds': frame_count / self.info['fps'], 'file': frame_filename, 'timestamp': datetime.datetime.now().isoformat() }) saved_count += 1 pbar.update(1) frame_count += 1 pbar.close() # Salvar metadados em CSV metadata_df = pd.DataFrame(metadata) metadata_df.to_csv(f"{frame_folder}/metadata.csv", index=False) # Criar arquivo ZIP zip_path = f"{Config.OUTPUT_DIR}/frames_{timestamp}.zip" shutil.make_archive(zip_path.replace('.zip', ''), 'zip', frame_folder) self.logger.log(f"Extraídos {saved_count} frames! ZIP: {zip_path}", "SUCCESS") # Baixar ZIP files.download(zip_path) return saved_count, metadata_df except Exception as e: self.logger.log(f"Erro na extração de frames: {str(e)}", "ERROR") return 0, None def create_timelapse(self, speed_factor=10): """Cria timelapse acelerando o vídeo""" self.logger.log(f"Criando timelapse (velocidade {speed_factor}x)...", "INFO") try: video_clip = VideoFileClip(self.video_path) new_fps = self.info['fps'] * speed_factor # Acelerar vídeo timelapse = video_clip.fx(lambda clip: clip.speedx(speed_factor)) output_file = f"{Config.OUTPUT_DIR}/timelapse_{speed_factor}x.mp4" timelapse.write_videofile(output_file, fps=min(new_fps, 60), codec='libx264', audio_codec='aac', verbose=False, logger=None) video_clip.close() file_size = os.path.getsize(output_file) / (1024*1024) self.logger.log(f"Timelapse criado! Tamanho: {file_size:.2f}MB", "SUCCESS") files.download(output_file) return output_file except Exception as e: self.logger.log(f"Erro ao criar timelapse: {str(e)}", "ERROR") return None def apply_filters(self, filter_type='grayscale'): """Aplica filtros ao vídeo completo""" self.logger.log(f"Aplicando filtro: {filter_type}", "INFO") try: video_clip = VideoFileClip(self.video_path) # Aplicar diferentes filtros if filter_type == 'grayscale': processed = video_clip.fx(lambda clip: clip.to_mask().to_video()) elif filter_type == 'sepia': def sepia(frame): frame = np.array(frame) sepia_matrix = np.array([[0.272, 0.534, 0.131], [0.349, 0.686, 0.168], [0.393, 0.769, 0.189]]) frame = np.dot(frame[...,:3], sepia_matrix.T) return np.clip(frame, 0, 255).astype(np.uint8) processed = video_clip.fl_image(sepia) elif filter_type == 'edge_detection': def edges(frame): gray = cv2.cvtColor(np.array(frame), cv2.COLOR_RGB2GRAY) edges = cv2.Canny(gray, 100, 200) return cv2.cvtColor(edges, cv2.COLOR_GRAY2RGB) processed = video_clip.fl_image(edges) else: processed = video_clip output_file = f"{Config.OUTPUT_DIR}/video_{filter_type}.mp4" processed.write_videofile(output_file, codec='libx264', audio_codec='aac', verbose=False, logger=None) video_clip.close() self.logger.log(f"Filtro '{filter_type}' aplicado!", "SUCCESS") files.download(output_file) return output_file except Exception as e: self.logger.log(f"Erro ao aplicar filtro: {str(e)}", "ERROR") return None def analyze_scene_changes(self, threshold=30): """Detecta mudanças de cena no vídeo""" self.logger.log("Analisando mudanças de cena...", "INFO") try: self.cap.set(cv2.CAP_PROP_POS_FRAMES, 0) prev_frame = None scene_changes = [] frame_num = 0 pbar = tqdm(total=self.info['total_frames'], desc="Analisando cenas") while True: ret, frame = self.cap.read() if not ret: break if prev_frame is not None: # Salvar gráfico chart_path = f"{Config.OUTPUT_DIR}/scene_changes.png" plt.savefig(chart_path, dpi=150, bbox_inches='tight') plt.show() # Salvar dados em JSON changes_path = f"{Config.OUTPUT_DIR}/scene_changes.json" with open(changes_path, 'w') as f: json.dump(scene_changes, f, indent=2) self.logger.log(f"Detectadas {len(scene_changes)} mudanças de cena", "SUCCESS") files.download(chart_path) files.download(changes_path) return scene_changes except Exception as e: self.logger.log(f"Erro na análise de cenas: {str(e)}", "ERROR") return [] def generate_statistics_report(self): """Gera relatório estatístico completo do vídeo""" self.logger.log("Gerando relatório estatístico...", "INFO") try: #
